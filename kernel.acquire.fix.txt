copy_package_payload() {
    local ROOT="$1"
    local PATHS="$2"
    local PATHNAME SRC_PATH

    while IFS= read -r PATHNAME; do
        [ -n "${PATHNAME}" ] || continue
        
        if [ -e "${ROOT}${PATHNAME}" ] || [ -L "${ROOT}${PATHNAME}" ]; then
            SRC_PATH="${ROOT}${PATHNAME}"
        elif [[ "${PATHNAME}" == "/usr/lib/"* ]] && { [ -e "${ROOT}/lib/${PATHNAME#/usr/lib/}" ] || [ -L "${ROOT}/lib/${PATHNAME#/usr/lib/}" ]; }; then
            SRC_PATH="${ROOT}/lib/${PATHNAME#/usr/lib/}"
        else
            continue
        fi

        mkdir -p "$(dirname "${PATHNAME}")"
        
        if [ -L "${SRC_PATH}" ]; then
            cp -a "${SRC_PATH}" "${PATHNAME}" 2>/dev/null || cp -L "${SRC_PATH}" "${PATHNAME}"
        else
            cp -a "${SRC_PATH}" "${PATHNAME}"
        fi
    done <"${PATHS}"
}

decompress_modules() {
    local FILE OUTPUT

    while IFS= read -r FILE; do
        [ -f "${FILE}" ] || continue
        
        OUTPUT="${FILE%.xz}"
        xz -dc "${FILE}" >"${OUTPUT}.tmp"
        chmod --reference="${FILE}" "${OUTPUT}.tmp"
        mv "${OUTPUT}.tmp" "${OUTPUT}"
        rm -f "${FILE}"
    done < <(find /lib/modules /usr/lib/modules -type f -name '*.ko.xz' 2>/dev/null)
    
    while IFS= read -r FILE; do
        [ -f "${FILE}" ] || continue
        
        OUTPUT="${FILE%.zst}"
        zstd -q -d -c "${FILE}" >"${OUTPUT}.tmp"
        chmod --reference="${FILE}" "${OUTPUT}.tmp"
        mv "${OUTPUT}.tmp" "${OUTPUT}"
        rm -f "${FILE}"
    done < <(find /lib/modules /usr/lib/modules -type f -name '*.ko.zst' 2>/dev/null)
    
    depmod -b / "${KERNEL_VERSION}"
}


classify_package() {
    local NAME="$1"
    local ROOT="$2"

    if [[ "${NAME}" =~ [-_]rt([-_]|[0-9]) ]] || [[ "${NAME}" =~ [-_]cloud([-_]|[0-9]) ]] || [[ "${NAME}" == *realtime* ]]; then
        printf '%s\n' dependency
        return
    fi

    if [ "${NAME}" = "${KERNEL_METAPACKAGE}" ]; then
        printf '%s\n' tracking-meta
    elif [[ "${NAME}" == linux-modules-extra-* ]]; then
        package_has "${ROOT}" '*/modules/*/*.ko*' || fail "modules-extra package has no modules: ${NAME}"
        printf '%s\n' modules-extra
    elif [[ "${NAME}" == linux-modules-* ]]; then
        package_has "${ROOT}" '*/modules/*/*.ko*' || fail "modules package has no modules: ${NAME}"
        printf '%s\n' modules
    elif [[ "${NAME}" == linux-binary-* ]]; then
        package_has "${ROOT}" '*/boot/vmlinuz-*' || package_has "${ROOT}" '*/usr/lib/modules/*/vmlinuz' || fail "binary package has no vmlinuz: ${NAME}"
        printf '%s\n' binary
    elif [[ "${NAME}" == linux-base-* ]]; then
        if package_has "${ROOT}" '*/boot/config-*' || package_has "${ROOT}" '*/usr/lib/modules/*/config'; then
            printf '%s\n' base
        else
            printf '%s\n' base-meta
        fi
    elif [[ "${NAME}" == linux-image-* ]]; then
        printf '%s\n' image
    else
        printf '%s\n' dependency
    fi
}

collect_package_paths() {
    local ROOT="$1"
    local OUTPUT="$2"
    local PATHNAME RELPATH

    : >"${OUTPUT}"
    
    while IFS= read -r PATHNAME; do
        [ -e "${PATHNAME}" ] || [ -L "${PATHNAME}" ] || continue
        
        case "${PATHNAME}" in
        */build | */source | */modules.alias | */modules.alias.bin | */modules.builtin.bin | \
            */modules.dep | */modules.dep.bin | */modules.devname | */modules.softdep | \
            */modules.symbols | */modules.symbols.bin | */modules.weakdep)
            continue
            ;;
        esac

        RELPATH="${PATHNAME#"${ROOT}"}"
        
        if [[ "${RELPATH}" == "/lib/modules/"* ]]; then
            RELPATH="/usr/lib/modules/${RELPATH#/lib/modules/}"
        elif [[ "${RELPATH}" == "/lib/"* ]] && [[ "${RELPATH}" != "/lib/modules/"* ]]; then
            RELPATH="/usr/lib/${RELPATH#/lib/}"
        fi
        
        printf '%s\n' "${RELPATH}" >>"${OUTPUT}"
    done < <(find "${ROOT}/boot" "${ROOT}/usr/lib/modules" "${ROOT}/lib/modules" "${ROOT}/lib" \( -type f -o -type l \) 2>/dev/null)

    sort -u -o "${OUTPUT}" "${OUTPUT}"
}

validate_kernel_closure() {
    local TRACKING IMAGE BINARY BASE MODULES VMLINUZ CONFIG SYSTEM_MAP KERNEL_PATH

    TRACKING="$(cut -f7 "${PACKAGE_TABLE}" | grep -c '^tracking-meta$' || true)"
    IMAGE="$(cut -f7 "${PACKAGE_TABLE}" | grep -c '^image$' || true)"
    BINARY="$(cut -f7 "${PACKAGE_TABLE}" | grep -c '^binary$' || true)"
    BASE="$(cut -f7 "${PACKAGE_TABLE}" | grep -c '^base$' || true)"
    MODULES="$(cut -f7 "${PACKAGE_TABLE}" | grep -c '^modules$' || true)"

    [ "${KERNEL_UPDATE_POLICY}" != track ] || [ "${TRACKING}" -eq 1 ] || fail "Tracked kernel requires one tracking metapackage"
    [ "${IMAGE}" -ge 1 ] || fail "Kernel closure requires at least one image package"

    if [ "${BINARY}" -gt 0 ]; then
        if [ "${MODULES}" -lt 1 ]; then
            fail "Debian split kernel closure is incomplete (missing modules)"
        fi
    fi

    mkdir -p /boot

    VMLINUZ="$(find /boot /usr/lib/modules /lib/modules -maxdepth 4 \( -type f -o -type l \) -name 'vmlinuz*' -print -quit 2>/dev/null)"
    [ -n "${VMLINUZ}" ] || fail "Kernel closure must provide a vmlinuz image"
    KERNEL_PATH="${VMLINUZ}"

    if [[ "${KERNEL_PATH}" == *"/modules/"* ]]; then
        KERNEL_VERSION="$(basename "$(dirname "${KERNEL_PATH}")")"
    else
        KERNEL_VERSION="${KERNEL_PATH##*/vmlinuz-}"
        if [ "${KERNEL_VERSION}" = "${KERNEL_PATH}" ] || [ -z "${KERNEL_VERSION}" ]; then
            KERNEL_VERSION="$(find /lib/modules /usr/lib/modules -mindepth 1 -maxdepth 1 -type d -exec basename {} \; 2>/dev/null | head -n 1)"
        fi
    fi

    [ -n "${KERNEL_VERSION}" ] || fail "Unable to determine kernel version from payload"

    CONFIG="/boot/config-${KERNEL_VERSION}"
    [ -f "${CONFIG}" ] || CONFIG="/usr/lib/modules/${KERNEL_VERSION}/config"
    [ -f "${CONFIG}" ] || CONFIG="/lib/modules/${KERNEL_VERSION}/config"
    [ -f "${CONFIG}" ] || fail "Kernel closure has no matching config: ${CONFIG}"

    SYSTEM_MAP="/boot/System.map-${KERNEL_VERSION}"
    [ -f "${SYSTEM_MAP}" ] || SYSTEM_MAP="/usr/lib/modules/${KERNEL_VERSION}/System.map"
    [ -f "${SYSTEM_MAP}" ] || SYSTEM_MAP="/lib/modules/${KERNEL_VERSION}/System.map"
    [ -f "${SYSTEM_MAP}" ] || fail "Kernel closure has no matching symbol map: ${SYSTEM_MAP}"

    [ -d "/lib/modules/${KERNEL_VERSION}" ] || [ -d "/usr/lib/modules/${KERNEL_VERSION}" ] ||
        fail "Kernel closure has no matching modules tree"

    grep -q '^CONFIG_EFI_STUB=y$' "${CONFIG}" || fail "Selected kernel lacks CONFIG_EFI_STUB=y"

    if [ "${DISTRIBUTION_ARCH}" = i386 ] && [ "${KERNEL_APT_ARCH}" = amd64 ]; then
        for SYMBOL in CONFIG_BINFMT_ELF CONFIG_IA32_EMULATION; do
            grep -q "^${SYMBOL}=y$" "${CONFIG}" || fail "Selected mixed-mode kernel lacks ${SYMBOL}=y"
        done
    fi

    if [ -f "${KERNEL_PATH}" ] && [ "${KERNEL_PATH}" != "/boot/vmlinuz-${KERNEL_VERSION}" ]; then
        cp -af "${KERNEL_PATH}" "/boot/vmlinuz-${KERNEL_VERSION}"
    fi

    if [ -f "${CONFIG}" ] && [ "${CONFIG}" != "/boot/config-${KERNEL_VERSION}" ]; then
        cp -af "${CONFIG}" "/boot/config-${KERNEL_VERSION}"
    fi

    if [ -f "${SYSTEM_MAP}" ] && [ "${SYSTEM_MAP}" != "/boot/System.map-${KERNEL_VERSION}" ]; then
        cp -af "${SYSTEM_MAP}" "/boot/System.map-${KERNEL_VERSION}"
    fi
}
